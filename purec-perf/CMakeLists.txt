cmake_minimum_required(VERSION 3.16)
project(zbperf VERSION 0.1.0 LANGUAGES C)
include(GNUInstallDirs)
include(CMakePackageConfigHelpers)
include(CTest)
find_package(Threads REQUIRED)
find_package(OpenSSL 1.1.1 REQUIRED)
find_package(PkgConfig REQUIRED)
pkg_check_modules(NGHTTP2 REQUIRED IMPORTED_TARGET libnghttp2>=1.40)

set(ZBPERF_SANITIZE "" CACHE STRING "Sanitizer: address,undefined or thread (empty disables)")
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
add_library(zbperf src/sdk.c src/queue.c src/pool.c src/proto.c)
add_library(zbperf::zbperf ALIAS zbperf)
target_include_directories(zbperf PUBLIC
    $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>
    $<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}>)
target_link_libraries(zbperf PRIVATE Threads::Threads OpenSSL::SSL OpenSSL::Crypto PkgConfig::NGHTTP2)
target_compile_options(zbperf PRIVATE -Wall -Wextra -Wpedantic -Werror)
set_target_properties(zbperf PROPERTIES VERSION ${PROJECT_VERSION} SOVERSION 0 POSITION_INDEPENDENT_CODE ON)
if(ZBPERF_SANITIZE)
    if(NOT ZBPERF_SANITIZE MATCHES "^(address|undefined|address,undefined|thread)$")
        message(FATAL_ERROR "Unsupported ZBPERF_SANITIZE")
    endif()
    target_compile_options(zbperf PUBLIC -fsanitize=${ZBPERF_SANITIZE} -fno-omit-frame-pointer)
    target_link_options(zbperf PUBLIC -fsanitize=${ZBPERF_SANITIZE})
endif()

if(BUILD_TESTING)
    foreach(test queue proto pool)
        add_executable(zbp_test_${test} tests/test_${test}.c src/${test}.c)
        target_include_directories(zbp_test_${test} PRIVATE include src)
        target_link_libraries(zbp_test_${test} PRIVATE Threads::Threads)
        target_compile_options(zbp_test_${test} PRIVATE -Wall -Wextra -Wpedantic -Werror -UNDEBUG)
        if(ZBPERF_SANITIZE)
            target_compile_options(zbp_test_${test} PRIVATE -fsanitize=${ZBPERF_SANITIZE} -fno-omit-frame-pointer)
            target_link_options(zbp_test_${test} PRIVATE -fsanitize=${ZBPERF_SANITIZE})
        endif()
        add_test(NAME ${test} COMMAND zbp_test_${test})
        set_tests_properties(${test} PROPERTIES TIMEOUT 60)
    endforeach()
    add_executable(zbp_test_state tests/test_state.c src/queue.c src/pool.c src/proto.c)
    target_include_directories(zbp_test_state PRIVATE include src)
    target_link_libraries(zbp_test_state PRIVATE Threads::Threads OpenSSL::SSL OpenSSL::Crypto PkgConfig::NGHTTP2)
    target_compile_options(zbp_test_state PRIVATE -Wall -Wextra -Wpedantic -Werror -UNDEBUG)
    if(ZBPERF_SANITIZE)
        target_compile_options(zbp_test_state PRIVATE -fsanitize=${ZBPERF_SANITIZE} -fno-omit-frame-pointer)
        target_link_options(zbp_test_state PRIVATE -fsanitize=${ZBPERF_SANITIZE})
    endif()
    add_test(NAME state COMMAND zbp_test_state)
    set_tests_properties(state PROPERTIES TIMEOUT 60)
    add_executable(zbp_integration_client tests/integration_client.c)
    target_link_libraries(zbp_integration_client PRIVATE zbperf::zbperf Threads::Threads)
    target_compile_options(zbp_integration_client PRIVATE -Wall -Wextra -Wpedantic -Werror -UNDEBUG)
endif()
add_executable(zbp_ingest_example examples/ingest.c)
target_link_libraries(zbp_ingest_example PRIVATE zbperf::zbperf)

configure_package_config_file(cmake/zbperfConfig.cmake.in "${CMAKE_CURRENT_BINARY_DIR}/zbperfConfig.cmake"
    INSTALL_DESTINATION "${CMAKE_INSTALL_LIBDIR}/cmake/zbperf")
write_basic_package_version_file("${CMAKE_CURRENT_BINARY_DIR}/zbperfConfigVersion.cmake"
    VERSION ${PROJECT_VERSION} COMPATIBILITY SameMajorVersion)
install(TARGETS zbperf EXPORT zbperfTargets
    ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR}
    LIBRARY DESTINATION ${CMAKE_INSTALL_LIBDIR}
    RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
install(FILES include/zbperf.h DESTINATION ${CMAKE_INSTALL_INCLUDEDIR})
install(EXPORT zbperfTargets NAMESPACE zbperf:: DESTINATION "${CMAKE_INSTALL_LIBDIR}/cmake/zbperf")
install(FILES "${CMAKE_CURRENT_BINARY_DIR}/zbperfConfig.cmake"
              "${CMAKE_CURRENT_BINARY_DIR}/zbperfConfigVersion.cmake"
    DESTINATION "${CMAKE_INSTALL_LIBDIR}/cmake/zbperf")
